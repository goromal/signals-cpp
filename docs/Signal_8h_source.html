410 BaseType xAt(const double& t) const
411 {
412 if (t == t_)
413 {
414 return x_;
415 }
416 else if (signalHistory_.size() > 0)
417 {
418 int idx = getInterpIndex(t);
419 if (idx < 0 || idx >= static_cast<int>(signalHistory_.size()))
420 {
421 switch (extrapolationMethod)
422 {
423 case ExtrapolationMethod::NANS:
424 return BaseSignalSpec::NansType();
425 break;
426 case ExtrapolationMethod::CLOSEST:
427 if (idx < 0)
428 {
429 return signalHistory_[0].x;
430 }
431 else
432 {
433 return signalHistory_[signalHistory_.size() - 1].x;
434 }
435 break;
436 case ExtrapolationMethod::ZEROS:
437 default:
438 return BaseSignalSpec::ZeroType();
439 break;
440 }
441 }
442 else
443 {
444 BaseType y = xAtIdx(idx);
445 TangentType dy;
446 switch (interpolationMethod)
447 {
448 case InterpolationMethod::ZERO_ORDER_HOLD: {
449 dy = TangentSignalSpec::ZeroType();
450 break;
451 }
452 case InterpolationMethod::LINEAR: {
453 double t1 = tAtIdx(idx);
454 double t2 = tAtIdx(idx + 1);
455 BaseType y1 = xAtIdx(idx);
456 BaseType y2 = xAtIdx(idx + 1);
457 dy = (t - t1) / (t2 - t1) * (y2 - y1);
458 break;
459 }
460 case InterpolationMethod::CUBIC_SPLINE: {
461 double t0 = tAtIdx(idx - 1);
462 double t1 = tAtIdx(idx);
463 double t2 = tAtIdx(idx + 1);
464 double t3 = tAtIdx(idx + 2);
465 BaseType y0 = xAtIdx(idx - 1);
466 BaseType y1 = xAtIdx(idx);
467 BaseType y2 = xAtIdx(idx + 1);
468 BaseType y3 = xAtIdx(idx + 2);
469 dy =
470 (t - t1) / (t2 - t1) *
471 ((y2 - y1) + (t2 - t) / (2. * (t2 - t1) * (t2 - t1)) *
472 (((t2 - t) * (t2 * (y1 - y0) + t0 * (y2 - y1) - t1 * (y2 - y0))) / (t1 - t0) +
473 ((t - t1) * (t3 * (y2 - y1) + t2 * (y3 - y1) - t1 * (y3 - y2))) / (t3 - t2)));
474 break;
475 }
476 }
477 return y + dy;
478 }
479 }
480 else
481 {
482 return BaseSignalSpec::ZeroType();
483 }

637 // Implementation note: should always be followed by a call to update()
638 bool calculateDerivative(const double& _t, const BaseType& _x, TangentType& _xdot)
639 {
640 const static double sigma = 0.05;
641 if (_t <= t_)
642 {
643 return false;
644 }
645 if (t_ >= 0)
646 {
647 const double dt = _t - t_;
648 const TangentType dx = _x - x_;
649 switch (derivativeMethod)
650 {
651 case DerivativeMethod::DIRTY:
652 _xdot = (2. * sigma - dt) / (2. * sigma + dt) * xdot_ + 2. / (2. * sigma + dt) * dx;
653 break;
654 case DerivativeMethod::FINITE_DIFF:
655 _xdot = dx / dt;
656 break;
657 }
658 }
659 else
660 {
661 _xdot = TangentSignalSpec::ZeroType();
662 }
663 return true;
664 }
